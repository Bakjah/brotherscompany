-- =============================================
-- 1. TABLE: members (MemberApp)
-- =============================================
DROP TABLE IF EXISTS `members`;
CREATE TABLE `members` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nama` VARCHAR(100) NOT NULL,
    `telepon` VARCHAR(20) NOT NULL,
    `compo_used` INT DEFAULT 0,
    `tanggal_daftar` DATE DEFAULT (CURRENT_DATE),
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- =============================================
-- 3. TABLE: sparepart_config (Sparepart Price)
-- =============================================

-- =============================================
-- 4. TABLE: transaksi (Calculator - Transaksi)
-- =============================================
DROP TABLE IF EXISTS `transaksi`;
CREATE TABLE `transaksi` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `member_id` INT NULL,
    `jenis_layanan` VARCHAR(50) NOT NULL,
    `compo_count` INT NOT NULL DEFAULT 0,
    `multiplier_used` DECIMAL(5,2) NOT NULL DEFAULT 0,
    `total_harga` DECIMAL(15,2) NOT NULL DEFAULT 0,
    `harga_sparepart` DECIMAL(15,2) NOT NULL DEFAULT 0,
    `harga_jasa` DECIMAL(15,2) NOT NULL DEFAULT 0,
    `compliment` VARCHAR(100) NULL,
    `tanggal` DATETIME DEFAULT CURRENT_TIMESTAMP,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (`member_id`) REFERENCES `members`(`id`) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================
-- 5. TABLE: laporan_mechanic (LaporanApp)
-- =============================================
DROP TABLE IF EXISTS `laporan_mechanic`;
CREATE TABLE `laporan_mechanic` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nama_mechanic` VARCHAR(100) NOT NULL,
    `compo_used` INT NOT NULL DEFAULT 0,
    `money_stored` DECIMAL(15,2) NOT NULL DEFAULT 0,
    `keterangan` TEXT NULL,
    `tanggal` DATE DEFAULT (CURRENT_DATE),
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =============================================
-- 6. TABLE: mecharing (Mecharing - Record Mechanic)
-- =============================================
DROP TABLE IF EXISTS `mecharing`;
CREATE TABLE `mecharing` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `nama_mechanic` VARCHAR(100) NOT NULL,
    `tanggal` DATE DEFAULT (CURRENT_DATE),
    `jam_masuk` TIME NULL,
    `jam_keluar` TIME NULL,
    `total_compo` INT DEFAULT 0,
    `total_transaksi` INT DEFAULT 0,
    `status` ENUM('active', 'inactive') DEFAULT 'active',
    `keterangan` TEXT NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Procedure untuk hitung harga
DELIMITER //
CREATE PROCEDURE `hitung_harga`(
    IN p_jenis_layanan VARCHAR(50),
    IN p_compo INT,
    OUT p_total DECIMAL(15,2),
    OUT p_sparepart DECIMAL(15,2),
    OUT p_jasa DECIMAL(15,2)
)
BEGIN
    DECLARE v_multiplier DECIMAL(5,2);
    DECLARE v_sparepart_price DECIMAL(10,2);

    SELECT `multiplier` INTO v_multiplier FROM `price_config` WHERE `jenis_layanan` = p_jenis_layanan;
    SELECT `harga_per_unit` INTO v_sparepart_price FROM `sparepart_config` LIMIT 1;

    SET p_sparepart = p_compo * v_sparepart_price;
    SET p_jasa = p_compo * v_multiplier;
    SET p_total = p_sparepart + p_jasa;
END //
DELIMITER ;
